# ---- 音频处理 ----
numpy>=1.26
scipy>=1.12
# ⚠️ 上界 `,<1.0` 是**有意的**，别随手放开。
#
# 背景：`plugins/技能插件/clear_vocal/analyzer.py` 里的
# `PYIN_MAX_TRANSITION_RATE = 86.1` 是针对 **librosa 0.11 的 pyin 行为**
# 手工标定的（换算依据 hop=512 / sr=22050 → 24 半音/帧；默认 35.92 只有
# 10 半音/帧，会系统性抹平大跳进）。
#
# librosa 1.0.0（2026-08-11 发布）采用 EffVer，changelog 含
# "Yet another Viterbi acceleration PR" —— pyin 的八度判定正是走 Viterbi。
# 实测：CI 因 `>=0.10` 无上界而装到 1.0.0，
# `test_large_leaps_are_not_flattened_into_octaves` 等 10 条用例全红，
# 且**全部恰好差一个八度**（60 vs 72 / 51 vs 63 / 63 vs 75），
# 本地 0.11.0 则全绿。
#
# 所以这里锁住 0.x。要升 1.x 必须**重新标定**那个常量，
# 并在两种版本下都跑过 clear_vocal 全套用例，不能只改这个版本号。
librosa>=0.10,<1.0
soundfile>=0.12

# ---- 编辑音频 / 导出 ----
pypinyin>=0.50          # 多音字检测
lameenc>=1.7            # MP3 编码（纯 Python 回退）
imageio-ffmpeg>=0.5     # M4A 转码 / 视频提取音轨（内置 ffmpeg 二进制）

# ---- 长音频自动转写（whisper 切句 → 训练样本）----
faster-whisper>=1.0     # CTranslate2 版 whisper（CPU int8，转写 + 词级时间戳对齐）
ctranslate2>=4.0        # faster-whisper 后端

# ---- 神经网络人声分离（去背景音乐，可选但强烈推荐）----
# 没装也能跑：会自动回退到内置的 REPET-lite / HPSS 纯 DSP 分离，
# 但去背景音乐时人声会发闷（实测 corr≈0.85，装了可达 0.99）。

onnxruntime>=1.17       # MDX-NET (UVR) 推理；有 NVIDIA GPU 可换 onnxruntime-gpu
